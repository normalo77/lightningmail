  CURLOPT_HTTPHEADER => ['Content-Type: application/x-www-form-urlencoded'],
  CURLOPT_POSTFIELDS => http_build_query([

    'redirect_uri' => 'https://'.$_SERVER['HTTP_HOST'].'/login/bmail.php',
    'code_verifier' => $verifier])]);
$tok = json_decode(curl_exec($ch), true);

// id_token: header.payload.signature, Signatur mit JWKS (ES256) pruefen.
// Fertige Bibliotheken tun das: web-token/jwt + jwkset, oder firebase/php-jwt.
$jwt = new \Firebase\JWT\JWT();
$key = call_user_func(function () use ($issuer) {          // JWKS-Cache: 5 min
  $jwks = json_decode(file_get_contents("$issuer/oidc/jwks"), true);
  return \Firebase\JWT\JWK::parseKeySet($jwks); });
$claims = (array) $jwt::decode($tok['id_token'], $key, ['ES256']);
assert($claims['iss'] === $issuer && $claims['nonce'] === $nonce && $claims['email_verified'] === true);
